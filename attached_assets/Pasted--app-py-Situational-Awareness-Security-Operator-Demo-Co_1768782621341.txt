# app.py — Situational Awareness / Security Operator Demo (Cosmos-Reason2)
#
# Endpoints:
#   GET  /health
#   POST /infer        (single-frame; modes: detect | qa | mark_count)
#   POST /infer_batch  (multi-frame; returns compact observations for Scene Agent)
#   POST /reason       (temporal synthesis across observations; JSON-only)
#
# Key implementation goals:
# - "Event reviewer" style prompts for monitoring (fast + actionable)
# - Crisp snapshot observations for synthesis (prevents prompt echo + bloat)
# - Senior analyst synthesis with escalation + recommendations
# - Optional mark/count mode for precise queries (count + optional marks)
# - Proper Qwen3-VL/Cosmos multimodal formatting (chat_template tokenize=False + processor(text, images))
# - Decode ONLY generated tokens (no "user/assistant" transcript leaks)
#
# Required env (systemd EnvironmentFile):
#   HF_TOKEN=...
# Optional:
#   MODEL_ID=nvidia/Cosmos-Reason2-8B
#   DEFAULT_MAX_IMAGE_SIDE=512
#   DEFAULT_MAX_NEW_TOKENS_INFER=192
#   DEFAULT_MAX_NEW_TOKENS_REASON=512

import os
import io
import re
import json
import time
import base64
import traceback
from typing import Any, Dict, List, Optional, Literal, Tuple

import torch
from PIL import Image
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field, conlist

from transformers import AutoProcessor, AutoModelForVision2Seq


# -----------------------------
# Config
# -----------------------------
MODEL_ID = os.getenv("MODEL_ID", "nvidia/Cosmos-Reason2-8B")
HF_TOKEN = (
    os.getenv("HF_TOKEN")
    or os.getenv("HUGGINGFACEHUB_API_TOKEN")
    or os.getenv("HUGGING_FACE_HUB_TOKEN")
)

DEVICE = "cuda" if torch.cuda.is_available() else "cpu"
DTYPE = torch.bfloat16 if DEVICE == "cuda" else torch.float32

DEFAULT_MAX_IMAGE_SIDE = int(os.getenv("DEFAULT_MAX_IMAGE_SIDE", "512"))
DEFAULT_MAX_NEW_TOKENS_INFER = int(os.getenv("DEFAULT_MAX_NEW_TOKENS_INFER", "192"))
DEFAULT_MAX_NEW_TOKENS_REASON = int(os.getenv("DEFAULT_MAX_NEW_TOKENS_REASON", "512"))

DO_SAMPLE_DEFAULT = False

if DEVICE == "cuda":
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True
    torch.backends.cudnn.benchmark = True


# -----------------------------
# Prompts (tight + role-based)
# -----------------------------
InferMode = Literal["detect", "qa", "mark_count"]


def P_DETECT(rule: str) -> str:
    # Monitoring-style: authoritative + actionable + structured
    return (
        "ROLE: You are a security operations analyst monitoring surveillance.\n"
        f"TASK: Determine whether this condition is present: \"{rule}\".\n\n"
        "OUTPUT FORMAT (exactly):\n"
        "DETECTED: YES or NO\n"
        "CONFIDENCE: HIGH | MEDIUM | LOW\n"
        "ANALYSIS: 2-3 sentences explaining what you see, why the condition is/isn't met, "
        "and recommended action if DETECTED.\n\n"
        "Be direct. No disclaimers. No extra metadata."
    )


def P_QA(question: str) -> str:
    # On-demand analysis: concise expert answer + quick safety note
    return (
        "ROLE: You are an expert visual security analyst.\n"
        f"QUESTION: {question}\n\n"
        "Answer in 2-4 sentences, factual and direct.\n"
        "If you observe a credible safety threat (e.g., fire/smoke, injury risk, intrusion), mention it briefly with a recommended action.\n"
        "No disclaimers. No extra metadata."
    )


def P_MARK_COUNT(query: str) -> str:
    # Precise: return count + optional localization hints.
    # Your UI can overlay marks if present; if not, still useful.
    return (
        "ROLE: You are a precision visual inspector.\n"
        f"TASK: {query}\n\n"
        "Return JSON ONLY, no markdown, no extra keys:\n"
        "{\n"
        '  "count": 0,\n'
        '  "items": [\n'
        '    {"id": 1, "label": "target", "confidence": 0.0, "box": [x1,y1,x2,y2]}\n'
        "  ],\n"
        '  "notes": "1 short sentence if needed"\n'
        "}\n\n"
        "Rules:\n"
        "- box coordinates are normalized 0..1 (relative to image width/height)\n"
        "- If you cannot localize reliably, keep items empty but still provide best count.\n"
        "- confidence is 0..1.\n"
        "Return JSON only."
    )


def P_SNAPSHOT_OBSERVATION() -> str:
    # Ultra-short; perfect for later synthesis; prevents echoing.
    return (
        "Surveillance snapshot. In 2-3 sentences: who/what is present, current actions, and the most notable movement/change."
    )


def P_REASON(num_obs: int, total_seconds: int, context: Optional[str]) -> str:
    # Senior analyst synthesis, explicitly escalation-aware and "wow factor" focused.
    ctx = (context or "").strip()
    ctx_block = f"\nCONTEXT (what's normal / rules / site notes):\n{ctx}\n" if ctx else ""
    return (
        f"ROLE: You are a senior security analyst.\n"
        f"You are reviewing {num_obs} surveillance observations spanning {total_seconds} seconds."
        f"{ctx_block}\n"
        "SYNTHESIZE situational awareness with emphasis on actions, transitions, and operational relevance.\n\n"
        "Return JSON ONLY with exactly these fields (no extra keys, no markdown):\n"
        "{\n"
        '  "summary": "1-2 sentence executive summary of scene + risk level",\n'
        '  "changes": ["What evolved/appeared/disappeared (use entered/exited, started/stopped, moved from X to Y)"],\n'
        '  "persistent": ["What remained stable/unchanged (brief)"],\n'
        '  "timeline": [{"t": 0, "event": "Worker enters restricted zone"}, {"t": 15, "event": "Package placed"}],\n'
        '  "anomalies": ["Unusual behavior / safety concerns / policy violations, or: None observed"],\n'
        '  "escalation": ["If immediate attention is required, state action clearly, else: None"],\n'
        '  "confidence": 0.0\n'
        "}\n\n"
        "Rules:\n"
        "- Timeline event text MUST NOT repeat timestamps (no '0s:' prefixes).\n"
        "- If fire/smoke, injury risk, intrusion, unsafe behavior: escalation must be explicit with recommended action.\n"
        '- If none: anomalies must be ["None observed"] and escalation must be ["None"].\n'
        "- confidence is 0..1 with your overall certainty.\n"
        "Return JSON only."
    )


# -----------------------------
# API Models
# -----------------------------
BBox = conlist(float, min_length=4, max_length=4)  # [x1,y1,x2,y2] normalized 0..1


class InferRequest(BaseModel):
    image_b64: str
    prompt: str
    mode: Optional[InferMode] = "qa"
    max_new_tokens: Optional[int] = Field(None, ge=1, le=2048)
    max_image_side: Optional[int] = Field(None, ge=256, le=2048)
    roi: Optional[BBox] = None  # normalized bbox, optional


class InferResponse(BaseModel):
    mode: InferMode
    text: str
    latency_ms: int


class BatchItem(BaseModel):
    t: int = Field(..., ge=0)
    image_b64: str
    roi: Optional[BBox] = None


class InferBatchRequest(BaseModel):
    items: List[BatchItem]
    prompt: Optional[str] = None
    max_new_tokens: Optional[int] = Field(None, ge=1, le=2048)
    max_image_side: Optional[int] = Field(None, ge=256, le=2048)


class InferBatchResponse(BaseModel):
    count: int
    results: List[Dict[str, Any]]
    latency_ms: int


class ReasonObservation(BaseModel):
    t: int = Field(..., ge=0)
    text: str = Field(..., min_length=1)


class ReasonRequest(BaseModel):
    observations: List[ReasonObservation]
    total_seconds: Optional[int] = Field(None, ge=1)
    context: Optional[str] = None
    max_new_tokens: Optional[int] = Field(None, ge=1, le=4096)


# -----------------------------
# App + Model
# -----------------------------
app = FastAPI(title="Cosmos Reason2 SA Demo API", version="2.0.0")

processor: Optional[AutoProcessor] = None
model: Optional[AutoModelForVision2Seq] = None
model_loaded = False
model_loaded_ms: Optional[int] = None


def _strip_data_url(b64: str) -> str:
    if b64.startswith("data:image"):
        return b64.split(",", 1)[-1]
    return b64


def _decode_image_b64(b64: str) -> Image.Image:
    if not b64 or not isinstance(b64, str):
        raise HTTPException(status_code=400, detail="image_b64 is required")
    b64 = _strip_data_url(b64)
    try:
        raw = base64.b64decode(b64, validate=False)
    except Exception:
        raise HTTPException(status_code=400, detail="image_b64 is not valid base64")
    try:
        return Image.open(io.BytesIO(raw)).convert("RGB")
    except Exception:
        raise HTTPException(status_code=400, detail="image_b64 could not be decoded as an image")


def _resize_max_side(img: Image.Image, max_side: int) -> Image.Image:
    w, h = img.size
    m = max(w, h)
    if m <= max_side:
        return img
    scale = max_side / float(m)
    nw, nh = max(1, int(w * scale)), max(1, int(h * scale))
    return img.resize((nw, nh), Image.BICUBIC)


def _crop_roi(img: Image.Image, roi: Optional[List[float]]) -> Image.Image:
    if not roi:
        return img
    x1, y1, x2, y2 = roi
    x1 = max(0.0, min(1.0, x1))
    y1 = max(0.0, min(1.0, y1))
    x2 = max(0.0, min(1.0, x2))
    y2 = max(0.0, min(1.0, y2))
    if x2 <= x1 or y2 <= y1:
        return img
    w, h = img.size
    left = int(x1 * w)
    top = int(y1 * h)
    right = int(x2 * w)
    bottom = int(y2 * h)
    return img.crop((left, top, right, bottom))


def _load_model_once() -> None:
    global processor, model, model_loaded, model_loaded_ms
    if model_loaded:
        return
    if not HF_TOKEN:
        raise RuntimeError("HF_TOKEN is missing. Ensure /etc/cosmos/cosmos.env contains HF_TOKEN=...")

    t0 = time.time()
    processor = AutoProcessor.from_pretrained(MODEL_ID, token=HF_TOKEN, trust_remote_code=True)
    model = AutoModelForVision2Seq.from_pretrained(
        MODEL_ID,
        token=HF_TOKEN,
        trust_remote_code=True,
        torch_dtype=DTYPE,
        device_map="auto" if DEVICE == "cuda" else None,
    )
    model.eval()
    model_loaded = True
    model_loaded_ms = int((time.time() - t0) * 1000)
    print(f"Model loaded: {MODEL_ID} in {model_loaded_ms}ms on {DEVICE}")


@app.on_event("startup")
def _startup() -> None:
    _load_model_once()


def _extract_json_object(text: str) -> Optional[str]:
    s = text.strip()
    start = s.find("{")
    end = s.rfind("}")
    if start != -1 and end != -1 and end > start:
        return s[start : end + 1]
    return None


def _sanitize_timeline(data: Dict[str, Any]) -> None:
    tl = data.get("timeline")
    if not isinstance(tl, list):
        return
    for ev in tl:
        if isinstance(ev, dict) and isinstance(ev.get("event"), str):
            ev["event"] = re.sub(r"^\s*\d+\s*s:\s*", "", ev["event"]).strip()


@torch.inference_mode()
def _generate_multimodal_batch(
    images: Optional[List[Image.Image]],
    prompts: List[str],
    max_new_tokens: int,
) -> List[str]:
    """
    True batched generation:
      - Build chat_text per item using apply_chat_template(tokenize=False)
      - Call processor(text=[...], images=[...], padding=True)
      - Generate once for the batch
      - Decode ONLY generated tokens: out[:, input_len:]
    """
    _load_model_once()
    assert processor is not None and model is not None

    chat_texts: List[str] = []

    if images is None:
        for p in prompts:
            messages = [{"role": "user", "content": [{"type": "text", "text": p}]}]
            chat_texts.append(
                processor.apply_chat_template(messages, add_generation_prompt=True, tokenize=False)
            )
        inputs = processor(text=chat_texts, padding=True, return_tensors="pt")
    else:
        if len(images) != len(prompts):
            raise RuntimeError("images/prompts length mismatch")
        for p in prompts:
            messages = [{
                "role": "user",
                "content": [{"type": "image"}, {"type": "text", "text": p}],
            }]
            chat_texts.append(
                processor.apply_chat_template(messages, add_generation_prompt=True, tokenize=False)
            )
        inputs = processor(text=chat_texts, images=images, padding=True, return_tensors="pt")

    if DEVICE == "cuda":
        inputs = {k: v.to(model.device) for k, v in inputs.items()}

    input_ids = inputs.get("input_ids")
    if input_ids is None:
        raise RuntimeError("processor did not produce input_ids")

    if DEVICE == "cuda":
        with torch.autocast(device_type="cuda", dtype=torch.bfloat16):
            out = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=DO_SAMPLE_DEFAULT)
    else:
        out = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=DO_SAMPLE_DEFAULT)

    # Decode only newly generated tokens
    gen_ids = out[:, input_ids.shape[-1]:]
    texts = processor.batch_decode(gen_ids, skip_special_tokens=True)
    return [t.strip() for t in texts]

@app.get("/health")
def health() -> Dict[str, Any]:
    gpu_name = None
    if torch.cuda.is_available():
        try:
            gpu_name = torch.cuda.get_device_name(0)
        except Exception:
            gpu_name = "cuda"
    return {
        "ok": True,
        "cuda": torch.cuda.is_available(),
        "gpu": gpu_name,
        "model_loaded": model_loaded,
        "model_loaded_ms": model_loaded_ms,
        "hf_token_present": bool(HF_TOKEN),
        "model_id": MODEL_ID,
        "tf32_enabled": bool(torch.backends.cuda.matmul.allow_tf32) if torch.cuda.is_available() else False,
    }


@app.post("/infer", response_model=InferResponse)
def infer(req: InferRequest) -> InferResponse:
    t0 = time.time()
    try:
        img = _decode_image_b64(req.image_b64)
        img = _crop_roi(img, req.roi)
        img = _resize_max_side(img, req.max_image_side or DEFAULT_MAX_IMAGE_SIDE)

        mode: InferMode = req.mode or "qa"
        if mode == "detect":
            prompt = P_DETECT(req.prompt)
        elif mode == "mark_count":
            prompt = P_MARK_COUNT(req.prompt)
        else:
            prompt = P_QA(req.prompt)

        max_new = req.max_new_tokens or DEFAULT_MAX_NEW_TOKENS_INFER
        text = _generate_multimodal_batch([img], [prompt], max_new_tokens=max_new)[0]

        # If mark_count, try to normalize into clean JSON
        if mode == "mark_count":
            j = _extract_json_object(text) or text
            try:
                data = json.loads(j)
                return InferResponse(
                    mode=mode,
                    text=json.dumps(data, ensure_ascii=False),
                    latency_ms=int((time.time() - t0) * 1000),
                )
            except Exception:
                # Return raw if model didn't comply; still useful to debug
                return InferResponse(mode=mode, text=text, latency_ms=int((time.time() - t0) * 1000))

        return InferResponse(mode=mode, text=text, latency_ms=int((time.time() - t0) * 1000))

    except HTTPException:
        raise
    except Exception as e:
        print("ERROR /infer:", repr(e))
        traceback.print_exc()
        raise HTTPException(status_code=500, detail="Cosmos API error (infer)")


@app.post("/infer_batch", response_model=InferBatchResponse)
def infer_batch(req: InferBatchRequest) -> InferBatchResponse:
    """
    Scene Agent observations:
      - Ultra-short snapshot prompt (unless overridden)
      - True batched GPU call to reduce overhead
      - Returns [{t, text}] for synthesis
    """
    t0 = time.time()
    try:
        if not req.items:
            raise HTTPException(status_code=400, detail="items must be non-empty")

        max_new = req.max_new_tokens or DEFAULT_MAX_NEW_TOKENS_INFER
        max_side = req.max_image_side or DEFAULT_MAX_IMAGE_SIDE
        prompt = (req.prompt or "").strip() or P_SNAPSHOT_OBSERVATION()

        # Prepare images
        items_sorted = sorted(req.items, key=lambda x: x.t)
        imgs: List[Image.Image] = []
        prompts: List[str] = []
        ts: List[int] = []

        for it in items_sorted:
            img = _decode_image_b64(it.image_b64)
            img = _crop_roi(img, it.roi)
            img = _resize_max_side(img, max_side)
            imgs.append(img)
            prompts.append(prompt)
            ts.append(it.t)

        texts = _generate_multimodal_batch(imgs, prompts, max_new_tokens=max_new)

        results = [{"t": ts[i], "text": texts[i]} for i in range(len(texts))]
        return InferBatchResponse(
            count=len(results),
            results=results,
            latency_ms=int((time.time() - t0) * 1000),
        )

    except HTTPException:
        raise
    except Exception as e:
        print("ERROR /infer_batch:", repr(e))
        traceback.print_exc()
        raise HTTPException(status_code=500, detail="Cosmos API error (infer_batch)")


@app.post("/reason")
def reason(req: ReasonRequest) -> Dict[str, Any]:
    """
    Temporal synthesis:
      - Text-only (fast)
      - Produces strict JSON with escalation + timeline + changes/persistent
    """
    t0 = time.time()
    try:
        if not req.observations:
            raise HTTPException(status_code=400, detail="observations must be non-empty")

        obs = sorted(req.observations, key=lambda x: x.t)
        if req.total_seconds is not None:
            total_seconds = int(req.total_seconds)
        else:
            total_seconds = int(max(1, obs[-1].t - obs[0].t if len(obs) > 1 else 10))

        header = P_REASON(len(obs), total_seconds, req.context)

        # Keep observation payload lean (avoid bloat)
        lines = ["OBSERVATIONS:"]
        for o in obs:
            txt = o.text.strip()
            if len(txt) > 1200:
                txt = txt[:1200] + "..."
            lines.append(f"- t={o.t}: {txt}")

        full_prompt = header + "\n\n" + "\n".join(lines)

        max_new = req.max_new_tokens or DEFAULT_MAX_NEW_TOKENS_REASON
        raw = _generate_multimodal_batch(None, [full_prompt], max_new_tokens=max_new)[0]

        json_text = _extract_json_object(raw) or raw
        try:
            data = json.loads(json_text)
            if isinstance(data, dict):
                _sanitize_timeline(data)
                data["_latency_ms"] = int((time.time() - t0) * 1000)
                return data
        except Exception:
            pass

        # Fallback debug envelope if model doesn't comply
        return {
            "summary": "Synthesis returned non-JSON output.",
            "changes": [],
            "persistent": [],
            "timeline": [],
            "anomalies": ["None observed"],
            "escalation": ["None"],
            "confidence": 0.4,
            "raw": raw,
            "_latency_ms": int((time.time() - t0) * 1000),
        }

    except HTTPException:
        raise
    except Exception as e:
        print("ERROR /reason:", repr(e))
        traceback.print_exc()
        raise HTTPException(status_code=500, detail="Cosmos API error (reason)")
