# app.py
import os
import base64
import io
from typing import Optional, Dict, Any

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from PIL import Image

import torch
from transformers import AutoProcessor, AutoModelForImageTextToText

MODEL_ID = "nvidia/Cosmos-Reason2-8B"

HF_TOKEN = os.environ.get("HF_TOKEN") or os.environ.get("HUGGINGFACE_HUB_TOKEN")

app = FastAPI(title="Cosmos Reason2 Inference API")

processor = None
model = None


# -------------------------
# Models
# -------------------------
class ROI(BaseModel):
    x: float
    y: float
    w: float
    h: float


class InferRequest(BaseModel):
    image_b64: str
    prompt: str
    roi: Optional[ROI] = None
    max_new_tokens: int = 96
    max_image_side: int = 512


# -------------------------
# Helpers
# -------------------------
def load_model():
    global processor, model

    if processor and model:
        return

    if not HF_TOKEN:
        raise HTTPException(
            status_code=401,
            detail="HF_TOKEN / HUGGINGFACE_HUB_TOKEN not set",
        )

    processor = AutoProcessor.from_pretrained(
        MODEL_ID,
        trust_remote_code=True,
        token=HF_TOKEN,
    )

    model = AutoModelForImageTextToText.from_pretrained(
        MODEL_ID,
        device_map="cuda",
        dtype=torch.bfloat16,
        trust_remote_code=True,
        token=HF_TOKEN,
    )
    model.eval()


def decode_image(b64: str) -> Image.Image:
    try:
        raw = base64.b64decode(b64, validate=True)
        return Image.open(io.BytesIO(raw)).convert("RGB")
    except Exception:
        raise HTTPException(status_code=400, detail="Invalid image_b64")


def crop_roi(img: Image.Image, roi: ROI) -> Image.Image:
    w, h = img.size
    x1 = int(roi.x * w)
    y1 = int(roi.y * h)
    x2 = int((roi.x + roi.w) * w)
    y2 = int((roi.y + roi.h) * h)
    return img.crop((x1, y1, x2, y2))


def resize_max_side(img: Image.Image, max_side: int) -> Image.Image:
    w, h = img.size
    scale = min(1.0, max_side / max(w, h))
    if scale < 1.0:
        img = img.resize((int(w * scale), int(h * scale)), Image.BICUBIC)
    return img


def clean_assistant(text: str) -> str:
    if "\nassistant\n" in text:
        return text.split("\nassistant\n", 1)[1].strip()
    return text.strip()


# -------------------------
# Endpoints
# -------------------------
@app.get("/health")
def health():
    return {
        "ok": True,
        "cuda": torch.cuda.is_available(),
        "gpu": torch.cuda.get_device_name(0) if torch.cuda.is_available() else None,
        "model_loaded": model is not None,
        "model": MODEL_ID,
    }


@app.post("/infer")
def infer(req: InferRequest) -> Dict[str, Any]:
    load_model()

    img = decode_image(req.image_b64)

    if req.roi:
        img = crop_roi(img, req.roi)

    img = resize_max_side(img, req.max_image_side)

    messages = [
        {
            "role": "user",
            "content": [
                {"type": "image"},
                {"type": "text", "text": req.prompt},
            ],
        }
    ]

    prompt = processor.apply_chat_template(
        messages,
        add_generation_prompt=True,
        tokenize=False,
    )

    inputs = processor(
        text=prompt,
        images=[img],
        return_tensors="pt",
    ).to("cuda")

    with torch.inference_mode():
        output = model.generate(
            **inputs,
            max_new_tokens=req.max_new_tokens,
            do_sample=False,
            temperature=0.0,
            num_beams=1,
        )

    text = processor.batch_decode(output, skip_special_tokens=True)[0]
    text = clean_assistant(text)

    return {"text": text}
