# app.py
# Cosmos-Reason2 FastAPI service (FINAL)
#
# Fixes:
# - Correct Qwen3-VL multimodal input formatting (prevents tokens:0/features mismatch)
#   Uses TWO-STEP pattern:
#     chat_text = processor.apply_chat_template(..., tokenize=False)
#     inputs = processor(text=[chat_text], images=[image], return_tensors="pt")
# - Updated prompts per your feedback:
#   /infer detect  : SOC analyst, structured output, recommended action
#   /infer qa      : expert visual analyst, concise 2-4 sentences, light safety note
#   /infer_batch   : ultra-short 2-3 sentences, movement/changes only (no lists)
#   /reason        : JSON-only synthesis, timeline event strings WITHOUT "0s:" prefix
#
# Endpoints:
#   GET  /health
#   POST /infer
#   POST /infer_batch
#   POST /reason
#
# Required env (via /etc/cosmos/cosmos.env):
#   HF_TOKEN=...
# Optional env:
#   MODEL_ID=nvidia/Cosmos-Reason2-8B
#   DEFAULT_MAX_IMAGE_SIDE=512
#   DEFAULT_MAX_NEW_TOKENS_INFER=192
#   DEFAULT_MAX_NEW_TOKENS_REASON=512

import os
import io
import re
import json
import time
import base64
import traceback
from typing import Any, Dict, List, Optional, Literal

import torch
from PIL import Image
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field

from transformers import AutoProcessor, AutoModelForVision2Seq


# -----------------------------
# Config
# -----------------------------
MODEL_ID = os.getenv("MODEL_ID", "nvidia/Cosmos-Reason2-8B")
HF_TOKEN = (
    os.getenv("HF_TOKEN")
    or os.getenv("HUGGINGFACEHUB_API_TOKEN")
    or os.getenv("HUGGING_FACE_HUB_TOKEN")
)

DEVICE = "cuda" if torch.cuda.is_available() else "cpu"
DTYPE = torch.bfloat16 if DEVICE == "cuda" else torch.float32

DEFAULT_MAX_IMAGE_SIDE = int(os.getenv("DEFAULT_MAX_IMAGE_SIDE", "512"))
DEFAULT_MAX_NEW_TOKENS_INFER = int(os.getenv("DEFAULT_MAX_NEW_TOKENS_INFER", "192"))
DEFAULT_MAX_NEW_TOKENS_REASON = int(os.getenv("DEFAULT_MAX_NEW_TOKENS_REASON", "512"))

# Deterministic for ops/UI
DEFAULT_DO_SAMPLE = False

# Small perf flags
if DEVICE == "cuda":
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True
    torch.backends.cudnn.benchmark = True


# -----------------------------
# Prompts (UPDATED)
# -----------------------------
InferMode = Literal["qa", "detect"]


def prompt_infer_detection(user_condition: str) -> str:
    return (
        "You are a security operations analyst monitoring live surveillance.\n"
        f'TASK: Evaluate if this condition is present: "{user_condition}".\n'
        "Examine the image carefully. Report in this exact format:\n"
        "DETECTED: YES or NO\n"
        "CONFIDENCE: HIGH | MEDIUM | LOW\n"
        "ANALYSIS: 2-3 sentences explaining what you see, why the condition is or isn't met, "
        "and recommended action if detected.\n"
        "Be direct. Do not include extra labels, disclaimers, or metadata."
    )


def prompt_infer_qa(user_question: str) -> str:
    return (
        "You are an expert visual analyst. Answer concisely and directly.\n"
        f"Question: {user_question}\n"
        "Provide a clear, factual response in 2-4 sentences. "
        "If you observe potential safety concerns, note them briefly.\n"
        "Do not include metadata or restate instructions."
    )


def prompt_batch_observation() -> str:
    return (
        "Surveillance snapshot. Report who/what is present, current actions, and notable movement or changes. "
        "2-3 sentences max. Do not include metadata or lists."
    )


def prompt_reason_synthesis(num_obs: int, total_seconds: Optional[int]) -> str:
    span_txt = f"spanning {total_seconds} seconds" if total_seconds is not None else "spanning an observation window"
    return (
        f"You are a senior security analyst reviewing {num_obs} surveillance snapshots {span_txt}.\n"
        "Synthesize a situational awareness report.\n\n"
        "Return JSON ONLY with exactly these fields (no extra keys, no markdown):\n"
        "{\n"
        '  "summary": "1-2 sentences describing the scene and activity level",\n'
        '  "timeline": [{"t": 0, "event": "Worker enters"}, {"t": 15, "event": "Package placed on conveyor"}],\n'
        '  "changes": ["What evolved, appeared, disappeared, or shifted"],\n'
        '  "anomalies": ["Unusual behavior, safety concerns, or policy violations (or: None observed)"],\n'
        '  "escalation": ["If immediate attention is required (fire, injury, intrusion, unsafe behavior), flag clearly with recommended action (or: None)"],\n'
        '  "confidence": 0.0\n'
        "}\n\n"
        "Guidelines:\n"
        "- Timeline events must be short/action-oriented and MUST NOT repeat timestamps inside the event text.\n"
        "- Changes should use transition language (entered/exited, started/stopped, moved from X to Y).\n"
        '- If there are no anomalies, set anomalies to ["None observed"].\n'
        '- If no escalation is needed, set escalation to ["None"].\n'
        "- confidence must be a number from 0.0 to 1.0.\n"
        "Return JSON only."
    )


# -----------------------------
# Request/Response Models
# -----------------------------
class InferRequest(BaseModel):
    image_b64: str = Field(..., description="Base64 image bytes. Data URL prefix is OK.")
    prompt: str = Field(..., description="Rule condition (detect) or user question (qa)")
    mode: Optional[InferMode] = Field(None, description="detect or qa (default qa)")
    max_new_tokens: Optional[int] = Field(None, ge=1, le=2048)
    max_image_side: Optional[int] = Field(None, ge=256, le=2048)


class InferResponse(BaseModel):
    text: str
    mode: InferMode
    latency_ms: int


class BatchItem(BaseModel):
    t: int = Field(..., ge=0)
    image_b64: str


class InferBatchRequest(BaseModel):
    items: List[BatchItem]
    prompt: Optional[str] = Field(None, description="Optional override prompt; else uses snapshot prompt")
    max_new_tokens: Optional[int] = Field(None, ge=1, le=2048)
    max_image_side: Optional[int] = Field(None, ge=256, le=2048)


class InferBatchResponse(BaseModel):
    count: int
    batched: bool = True
    results: List[Dict[str, Any]]
    latency_ms: int


class ReasonObservation(BaseModel):
    t: int = Field(..., ge=0)
    text: str = Field(..., min_length=1)


class ReasonRequest(BaseModel):
    observations: List[ReasonObservation]
    total_seconds: Optional[int] = Field(None, ge=1)
    max_new_tokens: Optional[int] = Field(None, ge=1, le=4096)


# -----------------------------
# App + Model
# -----------------------------
app = FastAPI(title="Cosmos Reason2 API", version="1.3.0")

processor: Optional[AutoProcessor] = None
model: Optional[AutoModelForVision2Seq] = None
model_loaded: bool = False
model_loaded_ms: Optional[int] = None


def _strip_data_url(b64: str) -> str:
    if b64.startswith("data:image"):
        return b64.split(",", 1)[-1]
    return b64


def _decode_image_b64(b64: str) -> Image.Image:
    if not b64 or not isinstance(b64, str):
        raise HTTPException(status_code=400, detail="image_b64 is required")
    b64 = _strip_data_url(b64)

    try:
        raw = base64.b64decode(b64, validate=False)
    except Exception:
        raise HTTPException(status_code=400, detail="image_b64 is not valid base64")

    try:
        img = Image.open(io.BytesIO(raw)).convert("RGB")
        return img
    except Exception:
        raise HTTPException(status_code=400, detail="image_b64 could not be decoded as an image")


def _resize_max_side(img: Image.Image, max_side: int) -> Image.Image:
    w, h = img.size
    m = max(w, h)
    if m <= max_side:
        return img
    scale = max_side / float(m)
    nw, nh = max(1, int(w * scale)), max(1, int(h * scale))
    return img.resize((nw, nh), Image.BICUBIC)


def _load_model_once() -> None:
    global processor, model, model_loaded, model_loaded_ms
    if model_loaded:
        return
    if not HF_TOKEN:
        raise RuntimeError("HF_TOKEN is not set in environment (/etc/cosmos/cosmos.env).")

    t0 = time.time()
    processor = AutoProcessor.from_pretrained(MODEL_ID, token=HF_TOKEN, trust_remote_code=True)
    model = AutoModelForVision2Seq.from_pretrained(
        MODEL_ID,
        token=HF_TOKEN,
        trust_remote_code=True,
        torch_dtype=DTYPE,
        device_map="auto" if DEVICE == "cuda" else None,
    )
    model.eval()
    model_loaded = True
    model_loaded_ms = int((time.time() - t0) * 1000)
    print(f"Model loaded: {MODEL_ID} in {model_loaded_ms}ms on {DEVICE}")


@app.on_event("startup")
def _startup() -> None:
    # Eager load for demos
    _load_model_once()


@torch.inference_mode()
def _generate_qwen(image: Optional[Image.Image], text_prompt: str, max_new_tokens: int) -> str:
    """
    Qwen3-VL/Cosmos input MUST include image placeholder tokens in the text stream.
    Correct approach:
      chat_text = processor.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
      inputs   = processor(text=[chat_text], images=[image], return_tensors="pt")
    """
    _load_model_once()
    assert processor is not None and model is not None

    if image is None:
        messages = [{"role": "user", "content": [{"type": "text", "text": text_prompt}]}]
        chat_text = processor.apply_chat_template(
            messages,
            add_generation_prompt=True,
            tokenize=False,   # <-- produce raw text WITH placeholders
        )
        inputs = processor(text=[chat_text], return_tensors="pt")
    else:
        messages = [{
            "role": "user",
            "content": [
                {"type": "image"},  # <-- placeholder; actual image goes in processor(images=...)
                {"type": "text", "text": text_prompt},
            ],
        }]
        chat_text = processor.apply_chat_template(
            messages,
            add_generation_prompt=True,
            tokenize=False,   # <-- produce raw text WITH placeholders
        )
        inputs = processor(text=[chat_text], images=[image], return_tensors="pt")

    if DEVICE == "cuda":
        inputs = {k: v.to(model.device) for k, v in inputs.items()}

    if DEVICE == "cuda":
        with torch.autocast(device_type="cuda", dtype=torch.bfloat16):
            out = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=DEFAULT_DO_SAMPLE)
    else:
        out = model.generate(**inputs, max_new_tokens=max_new_tokens, do_sample=DEFAULT_DO_SAMPLE)

    text = processor.batch_decode(out, skip_special_tokens=True)[0].strip()
    return text


def _clean_text_for_ui(s: str) -> str:
    return s.strip()


def _extract_first_json_object(text: str) -> Optional[str]:
    start = text.find("{")
    end = text.rfind("}")
    if start != -1 and end != -1 and end > start:
        return text[start : end + 1]
    return None


def _sanitize_timeline_events(data: Dict[str, Any]) -> None:
    tl = data.get("timeline")
    if not isinstance(tl, list):
        return
    for ev in tl:
        if isinstance(ev, dict) and isinstance(ev.get("event"), str):
            # Strip accidental "0s:" prefix if model adds it
            ev["event"] = re.sub(r"^\s*\d+\s*s:\s*", "", ev["event"]).strip()

@app.get("/health")
def health() -> Dict[str, Any]:
    gpu_name = None
    if torch.cuda.is_available():
        try:
            gpu_name = torch.cuda.get_device_name(0)
        except Exception:
            gpu_name = "cuda"

    return {
        "ok": True,
        "cuda": torch.cuda.is_available(),
        "gpu": gpu_name,
        "model_loaded": model_loaded,
        "model_loaded_ms": model_loaded_ms,
        "hf_token_present": bool(HF_TOKEN),
        "model_id": MODEL_ID,
        "tf32_enabled": bool(torch.backends.cuda.matmul.allow_tf32) if torch.cuda.is_available() else False,
    }


@app.post("/infer", response_model=InferResponse)
def infer(req: InferRequest) -> InferResponse:
    t0 = time.time()
    try:
        mode: InferMode = req.mode if req.mode in ("qa", "detect") else "qa"
        img = _decode_image_b64(req.image_b64)
        img = _resize_max_side(img, req.max_image_side or DEFAULT_MAX_IMAGE_SIDE)

        if mode == "detect":
            prompt = prompt_infer_detection(req.prompt)
        else:
            prompt = prompt_infer_qa(req.prompt)

        max_new = req.max_new_tokens or DEFAULT_MAX_NEW_TOKENS_INFER
        out = _generate_qwen(img, prompt, max_new)
        out = _clean_text_for_ui(out)

        return InferResponse(text=out, mode=mode, latency_ms=int((time.time() - t0) * 1000))
    except HTTPException:
        raise
    except Exception as e:
        print("ERROR /infer:", repr(e))
        traceback.print_exc()
        raise HTTPException(status_code=500, detail="Cosmos API error (infer)")


@app.post("/infer_batch", response_model=InferBatchResponse)
def infer_batch(req: InferBatchRequest) -> InferBatchResponse:
    t0 = time.time()
    try:
        if not req.items:
            raise HTTPException(status_code=400, detail="items must be non-empty")

        prompt = (req.prompt or "").strip() or prompt_batch_observation()
        max_new = req.max_new_tokens or DEFAULT_MAX_NEW_TOKENS_INFER
        max_side = req.max_image_side or DEFAULT_MAX_IMAGE_SIDE

        results: List[Dict[str, Any]] = []
        for it in req.items:
            img = _decode_image_b64(it.image_b64)
            img = _resize_max_side(img, max_side)
            txt = _generate_qwen(img, prompt, max_new)
            results.append({"t": it.t, "text": _clean_text_for_ui(txt)})

        return InferBatchResponse(count=len(results), results=results, latency_ms=int((time.time() - t0) * 1000))
    except HTTPException:
        raise
    except Exception as e:
        print("ERROR /infer_batch:", repr(e))
        traceback.print_exc()
        raise HTTPException(status_code=500, detail="Cosmos API error (infer_batch)")


@app.post("/reason")
def reason(req: ReasonRequest) -> Dict[str, Any]:
    t0 = time.time()
    try:
        if not req.observations:
            raise HTTPException(status_code=400, detail="observations must be non-empty")

        obs = sorted(req.observations, key=lambda x: x.t)
        inferred_total = obs[-1].t - obs[0].t if len(obs) > 1 else (req.total_seconds or 10)
        total_seconds = req.total_seconds or max(1, inferred_total)

        header = prompt_reason_synthesis(len(obs), int(total_seconds))

        lines = ["OBSERVATIONS:"]
        for o in obs:
            txt = o.text.strip()
            if len(txt) > 1400:
                txt = txt[:1400] + "..."
            lines.append(f"- t={o.t}: {txt}")

        prompt = header + "\n\n" + "\n".join(lines)
        max_new = req.max_new_tokens or DEFAULT_MAX_NEW_TOKENS_REASON

        # Text-only synthesis
        raw = _generate_qwen(None, prompt, max_new).strip()
        json_text = _extract_first_json_object(raw) or raw

        try:
            data = json.loads(json_text)
            if isinstance(data, dict):
                _sanitize_timeline_events(data)
                data["_latency_ms"] = int((time.time() - t0) * 1000)
            return data
        except Exception:
            return {
                "summary": "Synthesis returned non-JSON output.",
                "timeline": [],
                "changes": [],
                "anomalies": ["None observed"],
                "escalation": ["None"],
                "confidence": 0.4,
                "raw": raw,
                "_latency_ms": int((time.time() - t0) * 1000),
            }

    except HTTPException:
        raise
    except Exception as e:
        print("ERROR /reason:", repr(e))
        traceback.print_exc()
        raise HTTPException(status_code=500, detail="Cosmos API error (reason)")
